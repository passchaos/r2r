cmake_minimum_required(VERSION 3.16)
project(r2r_service_event_oracle LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(rmw_implementation REQUIRED)
find_package(std_msgs REQUIRED)
find_package(type_description_interfaces QUIET)
if(type_description_interfaces_FOUND AND NOT "$ENV{ROS_DISTRO}" STREQUAL "humble" AND NOT "$ENV{ROS_DISTRO}" STREQUAL "iron")
  add_executable(type_descriptions type_descriptions.cpp)
  target_compile_features(type_descriptions PRIVATE cxx_std_17)
  target_compile_options(type_descriptions PRIVATE -Wall -Wextra -Werror)
  ament_target_dependencies(type_descriptions rclcpp std_msgs type_description_interfaces rmw_implementation)
endif()
add_executable(qos_events qos_events.cpp)
target_compile_features(qos_events PRIVATE cxx_std_17)
target_compile_options(qos_events PRIVATE -Wall -Wextra -Werror)
ament_target_dependencies(qos_events rclcpp std_msgs rmw_implementation)
add_executable(service_events service_events.cpp)
target_compile_features(service_events PRIVATE cxx_std_17)
target_compile_options(service_events PRIVATE -Wall -Wextra -Werror)
ament_target_dependencies(service_events rclcpp example_interfaces rmw_implementation)
