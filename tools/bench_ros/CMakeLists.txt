cmake_minimum_required(VERSION 3.16)
project(r2r_ros_comparison LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rmw_implementation REQUIRED)
add_executable(rclcpp_comparison comparison.cpp)
target_compile_features(rclcpp_comparison PRIVATE cxx_std_17)
target_compile_options(rclcpp_comparison PRIVATE -O3 -DNDEBUG)
target_link_libraries(rclcpp_comparison ${CMAKE_DL_LIBS})
ament_target_dependencies(rclcpp_comparison rclcpp std_msgs rmw_implementation)
option(R2R_BUILD_STRING_BENCH "Build string-sequence codec comparison" OFF)
if(R2R_BUILD_STRING_BENCH)
  find_package(rcl_interfaces REQUIRED)
  add_executable(rclcpp_string_sequences string_sequences.cpp)
  target_compile_features(rclcpp_string_sequences PRIVATE cxx_std_17)
  target_compile_options(rclcpp_string_sequences PRIVATE -O3 -DNDEBUG)
  target_link_libraries(rclcpp_string_sequences ${CMAKE_DL_LIBS})
  ament_target_dependencies(rclcpp_string_sequences rclcpp rcl_interfaces rmw_implementation)
endif()
add_executable(rclcpp_roundtrip roundtrip.cpp)
target_compile_features(rclcpp_roundtrip PRIVATE cxx_std_17)
target_compile_options(rclcpp_roundtrip PRIVATE -O3 -DNDEBUG)
ament_target_dependencies(rclcpp_roundtrip rclcpp std_msgs rmw_implementation)
option(R2R_BUILD_SERVICE_BENCH "Build service roundtrip comparison" OFF)
option(R2R_BUILD_EXECUTOR_BENCH "Build multi-node service executor comparison" OFF)
if(R2R_BUILD_EXECUTOR_BENCH)
  find_package(example_interfaces REQUIRED)
  add_executable(rclcpp_multi_node_service multi_node_service.cpp)
  target_compile_features(rclcpp_multi_node_service PRIVATE cxx_std_17)
  target_compile_options(rclcpp_multi_node_service PRIVATE -O3 -DNDEBUG)
  ament_target_dependencies(rclcpp_multi_node_service rclcpp example_interfaces rmw_implementation)
endif()
if(R2R_BUILD_SERVICE_BENCH)
  find_package(example_interfaces REQUIRED)
  add_executable(rclcpp_service service.cpp)
  target_compile_features(rclcpp_service PRIVATE cxx_std_17)
  target_compile_options(rclcpp_service PRIVATE -O3 -DNDEBUG)
  ament_target_dependencies(rclcpp_service rclcpp example_interfaces rmw_implementation)
endif()
option(R2R_BUILD_ACTION_BENCH "Build action lifecycle comparison" OFF)
if(R2R_BUILD_ACTION_BENCH)
  find_package(example_interfaces REQUIRED)
  find_package(rclcpp_action REQUIRED)
  add_executable(rclcpp_action_roundtrip action.cpp)
  target_compile_features(rclcpp_action_roundtrip PRIVATE cxx_std_17)
  target_compile_options(rclcpp_action_roundtrip PRIVATE -O3 -DNDEBUG)
  target_link_libraries(rclcpp_action_roundtrip ${CMAKE_DL_LIBS})
  ament_target_dependencies(rclcpp_action_roundtrip rclcpp rclcpp_action example_interfaces rmw_implementation)
endif()
option(R2R_BUILD_ALLOCATION_PROBE "Build optional Linux/glibc allocation and resource probe" OFF)
if(R2R_BUILD_ALLOCATION_PROBE)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The allocation probe requires Linux/glibc")
  endif()
  enable_language(C)
  add_library(r2r_bench_probe SHARED probe/probe.c)
  target_compile_features(r2r_bench_probe PRIVATE c_std_11)
  target_compile_options(r2r_bench_probe PRIVATE -O2 -Wall -Wextra -Werror)
  target_link_libraries(r2r_bench_probe ${CMAKE_DL_LIBS} pthread)
endif()
