cmake_minimum_required(VERSION 3.16)
project(r2r_ros_comparison LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rmw_implementation REQUIRED)
add_executable(rclcpp_comparison comparison.cpp)
target_compile_features(rclcpp_comparison PRIVATE cxx_std_17)
target_compile_options(rclcpp_comparison PRIVATE -O3 -DNDEBUG)
ament_target_dependencies(rclcpp_comparison rclcpp std_msgs rmw_implementation)
add_executable(rclcpp_roundtrip roundtrip.cpp)
target_compile_features(rclcpp_roundtrip PRIVATE cxx_std_17)
target_compile_options(rclcpp_roundtrip PRIVATE -O3 -DNDEBUG)
ament_target_dependencies(rclcpp_roundtrip rclcpp std_msgs rmw_implementation)
option(R2R_BUILD_SERVICE_BENCH "Build service roundtrip comparison" OFF)
if(R2R_BUILD_SERVICE_BENCH)
  find_package(example_interfaces REQUIRED)
  add_executable(rclcpp_service service.cpp)
  target_compile_features(rclcpp_service PRIVATE cxx_std_17)
  target_compile_options(rclcpp_service PRIVATE -O3 -DNDEBUG)
  ament_target_dependencies(rclcpp_service rclcpp example_interfaces rmw_implementation)
endif()
option(R2R_BUILD_ACTION_BENCH "Build action lifecycle comparison" OFF)
if(R2R_BUILD_ACTION_BENCH)
  find_package(example_interfaces REQUIRED)
  find_package(rclcpp_action REQUIRED)
  add_executable(rclcpp_action_roundtrip action.cpp)
  target_compile_features(rclcpp_action_roundtrip PRIVATE cxx_std_17)
  target_compile_options(rclcpp_action_roundtrip PRIVATE -O3 -DNDEBUG)
  ament_target_dependencies(rclcpp_action_roundtrip rclcpp rclcpp_action example_interfaces rmw_implementation)
endif()
